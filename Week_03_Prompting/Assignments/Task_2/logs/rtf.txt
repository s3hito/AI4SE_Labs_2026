def occurrences(text, values):
    found=[]
    for value in sorted(values,key=len,reverse=True):
        start=0
        while True:
            i=text.find(value,start)
            if i<0: break
            found.append((i,value)); start=i+1
    return found

def city_for(q, end=None):
    text=q.lower() if end is None else q.lower()[:end]
    occ=occurrences(text,cities)
    return max(occ,key=lambda x:x[0])[1] if occ else None

def food_for(q, city_pos=None):
    text=q.lower()
    occ=occurrences(text,foods)
    if not occ: return None
    if city_pos is None: return max(occ,key=lambda x:x[0])[1]
    before=[x for x in occ if x[0] < city_pos]
    after=[x for x in occ if x[0] >= city_pos]
    # The intended food is adjacent to the target city; injected menu words are farther away.
    if before: return min(before,key=lambda x:city_pos-(x[0]+len(x[1])))[1]
    return min(after,key=lambda x:x[0]-city_pos)[1]
